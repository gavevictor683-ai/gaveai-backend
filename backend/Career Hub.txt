    window.saveResumeAndCoverLetter = async () => {
      const user = auth.currentUser;
      if (!user) return;

      if (resumeUploading) {
        showToast(
          "Please wait until resume upload finishes.",
          "error"
        );
        return;
      }

      const resumeInput =
        document.getElementById(
          "edit-resume"
        );

      let resumeURL = null;

      const saveBtn =
        document.getElementById(
          "gmt-save-resume-btn"
        );

      if (
        resumeInput.files &&
        resumeInput.files[0]
      ) {
        const file =
          resumeInput.files[0];

        if (
          file.type !==
          "application/pdf"
        ) {
          showToast(
            "Please upload a valid PDF resume file",
            "error"
          );
          return;
        }

        resumeUploading = true;

        saveBtn.disabled = true;
        saveBtn.innerText =
          "Uploading Resume...";

        showToast(
          "Uploading resume PDF to ImageKit..."
        );

        try {
          const formData =
            new FormData();

          formData.append(
            "file",
            file
          );

          formData.append(
            "userId",
            user.uid
          );

          const response =
            await fetch(
              "https://gaveai-backend.onrender.com/upload-resume",
              {
                method: "POST",
                body: formData
              }
            );

          const result =
            await response.json();

          if (
            !response.ok ||
            !result.success
          ) {
            throw new Error(
              result.error ||
              "Failed to upload resume"
            );
          }

          resumeURL =
            result.url;

        } catch (err) {
          console.error(
            "ImageKit resume upload error:",
            err
          );

          showToast(
            "Upload failed: " +
              err.message,
            "error"
          );

          resumeUploading = false;

          saveBtn.disabled = false;
          saveBtn.innerText =
            "Save Changes";

          return;

        } finally {
          resumeUploading = false;
        }
      }

      const updatedData = {
        coverLetter:
          document
            .getElementById(
              "edit-cover-letter"
            )
            .value.trim()
      };

      if (resumeURL) {
        updatedData.resumeURL =
          resumeURL;

        updatedData.resumeUpdatedAt =
          new Date().toISOString();
      }

      try {
        saveBtn.innerText =
          "Saving...";

        await updateDoc(
          doc(
            db,
            "users",
            user.uid
          ),
          updatedData
        );

        showToast(
          "Resume & Cover Letter updated successfully!"
        );

        closeResumeModal();

        location.reload();

      } catch (error) {
        console.error(
          "Firestore update error:",
          error
        );

        showToast(
          "Error updating: " +
            error.message,
          "error"
        );

        saveBtn.disabled = false;
        saveBtn.innerText =
          "Save Changes";
      }
    };

              ? userDoc.data().skills

            }

        );

            e.message,
          "error"
        );
      }

    window.gmtAddCertification = async () => {
      const nameInput =
        document.getElementById(
          "gmt-certification-name"
        );

      const issuerInput =
        document.getElementById(
          "gmt-certification-issuer"
        );

      const fileInput =
        document.getElementById(
          "gmt-certification-file"
        );

      const uploadBtn =
        document.getElementById(
          "gmt-upload-cert-btn"
        );

      const currentUser =
        auth.currentUser;

      if (
        !nameInput ||
        !issuerInput ||
        !fileInput
      ) {
        showToast(
          "Certificate form is missing.",
          "error"
        );
        return;
      }

      const name =
        nameInput.value.trim();

      const issuer =
        issuerInput.value.trim();

      if (!name) {
        showToast(
          "Please enter the certification name.",
          "error"
        );
        return;
      }

      if (
        !fileInput.files ||
        !fileInput.files[0]
      ) {
        showToast(
          "Please choose a Certificate PDF file.",
          "error"
        );
        return;
      }

      const file =
        fileInput.files[0];

      if (
        file.type !==
        "application/pdf"
      ) {
        showToast(
          "Only PDF files are allowed for certificates.",
          "error"
        );
        return;
      }

      if (!currentUser) {
        showToast(
          "You must be signed in to upload a certificate.",
          "error"
        );
        return;
      }

      const formData =
        new FormData();

      formData.append(
        "file",
        file
      );

      formData.append(
        "userId",
        currentUser.uid
      );

      try {
        if (uploadBtn) {
          uploadBtn.disabled =
            true;

          uploadBtn.innerText =
            "Uploading Certificate...";
        }

        showToast(
          "Uploading certificate PDF to ImageKit..."
        );

        const response =
          await fetch(
            "https://gaveai-backend.onrender.com/upload-certificate",
            {
              method: "POST",
              body: formData
            }
          );

        const result =
          await response.json();

        if (
          !response.ok ||
          !result.success ||
          !result.url
        ) {
          throw new Error(
            result.error ||
              "Failed to upload certificate"
          );
        }

        const newCert = {
          name: name,
          issuingOrganization:
            issuer,
          url: result.url,
          uploadedAt:

    window.gmtStartDirectMessage =
      async (
        targetUserId,
        targetName,
        targetRole
      ) => {
        const currentUser =
          auth.currentUser;

        if (!currentUser) {
          showToast(
            "Please sign in to message users.",
            "error"
          );
          return;
        }

        if (
          currentUser.uid ===
          targetUserId
        ) {
          showToast(
            "You cannot message yourself.",
            "error"
          );
          return;
        }

        const conversationId =
          [
            currentUser.uid,
            targetUserId
          ]
            .sort()
            .join("_");

        try {
          const convRef =
            doc(
              db,
              "conversations",
              conversationId
            );

          const convSnap =
            await getDoc(
              convRef
            );

          const userDoc =
            await getDoc(
              doc(
                db,
                "users",
                currentUser.uid
              )
            );

          const currentName =
            userDoc.exists()
              ? (
                  userDoc.data()
                    .fullName ||
                  "User"
                )
              : "User";

          if (!convSnap.exists()) {
            await setDoc(
              convRef,
              {
                participants: [
                  currentUser.uid,
                  targetUserId
                ],

                participantNames: [
                  currentName,
                  targetName
                ],

                lastMessage:
                  "Conversation started",

                lastMessageAt:
                  serverTimestamp(),

                createdAt:
                  serverTimestamp()
              }
            );
          }

          triggerFirebaseAction(
            "openChat"
          );

          setTimeout(
            () => {
              gmtSelectConversation(
                conversationId,
                targetName
              );
            },
            300
          );

        } catch (e) {
          console.error(
            "Error starting chat:",
            e
          );

          showToast(
            "Error starting chat: " +
              e.message,
            "error"
          );
        }
      };


    async function initMessagingUI() {
      const currentUser =
        auth.currentUser;

      if (!currentUser) return;

      const convContainer =
        document.getElementById(
          "gmt-conversations-container"
        );

      if (!convContainer) return;

      const q =
        query(
          collection(
            db,
            "conversations"
          ),
          where(
            "participants",
            "array-contains",
            currentUser.uid
          )
        );

      if (activeUnsubscribe) {
        activeUnsubscribe();
        activeUnsubscribe = null;
      }

      activeUnsubscribe =
        onSnapshot(
          q,
          snapshot => {
            if (snapshot.empty) {
              convContainer.innerHTML =
                `<div class="gmt-empty-state" style="padding:10px;font-size:13px;">
                  No conversations yet.
                </div>`;

              return;
            }

            let html = "";

            snapshot.forEach(
              docSnap => {
                const data =
                  docSnap.data();

                const convId =
                  docSnap.id;

                const names =
                  data.participantNames ||
                  [];

                const otherName =
                  names.find(
                    n =>
                      n !==
                        currentUser.displayName &&
                      n !== "User"
                  ) ||
                  "Chat Partner";

                const isActive =
                  currentActiveConversationId ===
                  convId
                    ? "active"
                    : "";

                const safeOtherName =
                  otherName.replace(
                    /'/g,
                    "\\'"
                  );

                html += `
                  <div
                    class="gmt-conversation-item ${isActive}"
                    onclick="gmtSelectConversation(
                      '${convId}',
                      '${safeOtherName}'
                    )"
                  >

                    <strong>
                      ${otherName}
                    </strong>

                    <div
                      style="
                        font-size:11px;
                        color:#666;
                        white-space:nowrap;
                        overflow:hidden;
                        text-overflow:ellipsis;
                      "
                    >
                      ${
                        data.lastMessage ||
                        "No messages yet"
                      }
                    </div>

                  </div>
                `;
              }
            );

            convContainer.innerHTML =
              html;
          }
        );
    }

        partnerName
      ) => {
        currentActiveConversationId =
          convId;

        document
          .querySelectorAll(
            ".gmt-conversation-item"
          )
          .forEach(
            el =>
              el.classList.remove(
                "active"
              )
          );

        const msgInput =
          document.getElementById(
            "gmt-message-input"
          );

        const sendBtn =
          document.getElementById(
            "gmt-send-msg-btn"
          );

        if (msgInput) {
          msgInput.disabled =
            false;
        }

        if (sendBtn) {
          sendBtn.disabled =
            false;
        }

        const msgContainer =
          document.getElementById(
            "gmt-messages-container"
          );

        const messagesQuery =
          query(
            collection(
              db,
              "conversations",
              convId,
              "messages"
            ),
            orderBy(
              "createdAt",
              "asc"
            )
          );

        if (
          activeMessagesUnsubscribe
        ) {
          activeMessagesUnsubscribe();
          activeMessagesUnsubscribe =
            null;
        }

        activeMessagesUnsubscribe =
          onSnapshot(
            messagesQuery,
            snapshot => {
              let msgHtml = "";

              const currentUser =
                auth.currentUser;

              snapshot.forEach(
                docSnap => {
                  const msg =
                    docSnap.data();

                  const isSent =
                    msg.senderId ===
                    currentUser.uid;

                  msgHtml += `
                    <div
                      class="gmt-message-bubble ${
                        isSent
                          ? "sent"
                          : "received"
                      }"
                    >

                      <div
                        style="
                          font-size:10px;
                          font-weight:bold;
                          margin-bottom:2px;
                          color:#444;
                        "
                      >
                        ${
                          msg.senderName ||
                          "User"
                        }
                      </div>

                      <div>
                        ${msg.text || ""}
                      </div>

                    </div>
                  `;
                }
              );

              if (msgContainer) {
                msgContainer.innerHTML =
                  msgHtml ||
                  `<div class="gmt-empty-state" style="padding:20px;">
                    No messages in this chat yet.
                  </div>`;

                msgContainer.scrollTop =
                  msgContainer.scrollHeight;
              }
            }
          );
      };


    window.gmtSendMessage =
      async () => {
        const input =
          document.getElementById(
            "gmt-message-input"
          );

        const currentUser =
          auth.currentUser;

        if (
          !input ||
          !currentUser ||
          !currentActiveConversationId
        ) {
          return;
        }

        const text =
          input.value.trim();

        if (!text) return;

        try {
          input.value = "";

          const userDoc =
            await getDoc(
              doc(
                db,
                "users",
                currentUser.uid
              )
            );

          const senderName =
            userDoc.exists()
              ? (
                  userDoc.data()
                    .fullName ||
                  "User"
                )
              : "User";

          const senderRole =
            userDoc.exists()
              ? (
                  userDoc.data()
                    .role ||
                  "worker"
                )
              : "worker";

          await addDoc(
            collection(
              db,
              "conversations",
              currentActiveConversationId,
              "messages"
            ),
            {
              senderId:
                currentUser.uid,
              senderName:
                senderName,
              senderRole:
                senderRole,
              text:
                text,
              createdAt:
                serverTimestamp()
            }
          );

          await updateDoc(
            doc(
              db,
              "conversations",
              currentActiveConversationId
            ),
            {
              lastMessage:
                text,
              lastMessageAt:
                serverTimestamp()
            }
          );

        } catch (e) {
          console.error(
            "Error sending message:",
            e
          );

          showToast(
            "Error sending message: " +
              e.message,
            "error"
          );
        }
      };
  </script>
