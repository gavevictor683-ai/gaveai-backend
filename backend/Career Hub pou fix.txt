  async function uploadResumeFile(file, currentUser) {
    const storageRef = ref(storage, `resumes/${currentUser.uid}/${file.name}`);
    await uploadBytes(storageRef, file);
    const downloadURL = await getDownloadURL(storageRef);
    return downloadURL;
  }

  window.saveResumeAndCoverLetter = async () => {
    const user = auth.currentUser;
    if (!user) return;

    if (resumeUploading) {
      showToast("Please wait until resume upload finishes.", "error");
      return;
    }

    const resumeInput = document.getElementById('edit-resume');
    let resumeURL = null;
    const saveBtn = document.getElementById('gmt-save-resume-btn');

    if (resumeInput.files && resumeInput.files[0]) {
      const file = resumeInput.files[0];
      if (file.type !== "application/pdf") {
        showToast("Please upload a valid PDF resume file", "error");
        return;
      }

      resumeUploading = true;
      saveBtn.disabled = true;
      saveBtn.innerText = "Uploading Resume...";
      showToast("Uploading resume PDF to Firebase...");

      try {
        resumeURL = await uploadResumeFile(file, user);
      } catch (err) {
        console.error("Firebase Storage upload error:", err);
        showToast("Upload failed: " + err.message, "error");
        resumeUploading = false;
        saveBtn.disabled = false;
        saveBtn.innerText = "Save Changes";
        return;
      } finally {
        resumeUploading = false;
      }
    }

    const updatedData = {
      coverLetter: document.getElementById('edit-cover-letter').value.trim()
    };

    if (resumeURL) {
      updatedData.resumeURL = resumeURL;
      updatedData.resumeUpdatedAt = new Date().toISOString();
    }

    try {
      saveBtn.innerText = "Saving...";
      await updateDoc(doc(db, "users", user.uid), updatedData);
      showToast("Resume & Cover Letter updated successfully!");
      closeResumeModal();
      location.reload();
    } catch (error) {
      console.error("Firestore update error:", error);
      showToast("Error updating: " + error.message, "error");
      saveBtn.disabled = false;
      saveBtn.innerText = "Save Changes";
    }
  };

  window.gmtAddCertification = async () => {
    const nameInput = document.getElementById("gmt-certification-name");
    const issuerInput = document.getElementById("gmt-certification-issuer");
    const fileInput = document.getElementById("gmt-certification-file");
    const uploadBtn = document.getElementById("gmt-upload-cert-btn");
    const currentUser = auth.currentUser;

    if (!nameInput || !issuerInput) return;
    const name = nameInput.value.trim();
    const issuer = issuerInput.value.trim();

    if (!name) {
      showToast("Please enter the certification name.", "error");
      return;
    }

    if (!fileInput || !fileInput.files || !fileInput.files[0]) {
      showToast("Please choose a Certificate PDF file.", "error");
      return;
    }

    const file = fileInput.files[0];
    if (file.type !== "application/pdf") {
      showToast("Only PDF files are allowed for certificates.", "error");
      return;
    }

    if (!currentUser) {
      showToast("You must be signed in to upload a certificate.", "error");
      return;
    }

    const formData = new FormData();
    formData.append("file", file);
    formData.append("userId", currentUser.uid);

    try {
      uploadBtn.disabled = true;
      uploadBtn.innerText = "Uploading Certificate...";
      showToast("Uploading certificate PDF to ImageKit...");

      const response = await fetch("/upload-certificate", {
        method: "POST",
        body: formData
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        throw new Error(result.error || "Failed to upload certificate");
      }

      const list = document.getElementById("gmt-certification-list");
      if (list) {
        const certification = document.createElement("div");
        certification.className = "gmt-certification-item";
        certification.innerHTML = `
          <strong>${name}</strong>
          ${issuer ? `<div>${issuer}</div>` : ""}
          <div><a href="${result.url}" target="_blank" style="color: #0a66c2; font-weight:600; text-decoration:none;">📄 View Certificate PDF</a></div>
        `;
        list.appendChild(certification);
      }

      nameInput.value = "";
      issuerInput.value = "";
      fileInput.value = "";
      showToast("Certification uploaded and saved successfully!");
    } catch (err) {
      console.error("Certificate upload error:", err);
      showToast("Error uploading certificate: " + err.message, "error");
    } finally {
      uploadBtn.disabled = false;
      uploadBtn.innerText = "+ Add Certification";
    }
  };

  window.gmtSendMessage = () => {
    const input = document.getElementById("gmt-message-input");
    if (!input) return;
    const message = input.value.trim();
    if (!message) {
      showToast("Write a message first.", "error");
      return;
    }
    showToast("Messaging is ready to be connected to Firestore.");
    input.value = "";
  };
