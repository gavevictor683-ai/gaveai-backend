<script type="module">
  import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js";
  import { 
    getAuth, 
    signInWithEmailAndPassword, 
    createUserWithEmailAndPassword, 
    onAuthStateChanged, 
    signOut, 
    sendPasswordResetEmail,
    sendEmailVerification,
    GoogleAuthProvider,
    signInWithPopup,
    setPersistence,
    browserLocalPersistence,
    browserSessionPersistence
  } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-auth.js";
  import { 
    getFirestore, 
    doc, 
    setDoc, 
    getDoc, 
    updateDoc 
  } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";

  const firebaseConfig = {
    apiKey: "AIzaSyDtplo33r-gef8cV1nzvpHNR7XEuP-AIt8",
    authDomain: "gave-money-tips.firebaseapp.com",
    projectId: "gave-money-tips",
    storageBucket: "gave-money-tips.firebasestorage.app",
    messagingSenderId: "402925165043",
    appId: "1:402925165043:web:db3fc0313a98aa86999a86",
    measurementId: "G-CNHBFV81GS"
  };

  const app = initializeApp(firebaseConfig);
  const auth = getAuth(app);
  const db = getFirestore(app);

  function showToast(message, type = 'success') {
    const container = document.getElementById('gmt-toast-container');
    if (!container) return;

    const toast = document.createElement('div');
    toast.className = `gmt-toast ${type}`;
    
    const icon = type === 'success' ? '🟢' : '❌';
    toast.innerHTML = `<span>${icon}</span> <div>${message}</div>`;

    container.appendChild(toast);

    setTimeout(() => {
      toast.remove();
    }, 5000);
  }

  window.registerUser = async () => {
    const fullName = document.getElementById('auth-name').value.trim();
    const country = document.getElementById('auth-country').value.trim();
    const email = document.getElementById('auth-email').value.trim();
    const password = document.getElementById('auth-password').value;
    const role = document.getElementById('register-role').value;
    const rememberMe = document.getElementById('remember-me').checked;

    if (!fullName || !country || !email || !password) {
      showToast("Please fill out Full Name, Country, Email, and Password!", "error");
      return;
    }

    try {
      await setPersistence(auth, rememberMe ? browserLocalPersistence : browserSessionPersistence);

      const userCredential = await createUserWithEmailAndPassword(auth, email, password);
      const user = userCredential.user;

      await setDoc(doc(db, "users", user.uid), {
        fullName: fullName,
        country: country,
        city: "",
        languages: "",
        skills: "",
        experience: "",
        availability: role === 'worker' ? "OPEN TO WORK" : "HIRING",
        timezone: "",
        linkedin: "",
        portfolio: "",
        bio: "",
        email: email,
        role: role,
        createdAt: new Date().toISOString()
      });

      await sendEmailVerification(user);
      showToast("Account created successfully!<br>📧 Check your Inbox or Spam folder for the verification email.");
    } catch (error) {
      showToast("Registration error: " + error.message, "error");
    }
  };

  window.loginUser = async () => {
    const email = document.getElementById('auth-email').value.trim();
    const password = document.getElementById('auth-password').value;
    const rememberMe = document.getElementById('remember-me').checked;

    if (!email || !password) {
      showToast("Please enter your email and password!", "error");
      return;
    }

    try {
      await setPersistence(auth, rememberMe ? browserLocalPersistence : browserSessionPersistence);
      await signInWithEmailAndPassword(auth, email, password);
      showToast("Signed in successfully!");
    } catch (error) {
      showToast("Login error: " + error.message, "error");
    }
  };

  window.signInWithGoogle = async () => {
    const provider = new GoogleAuthProvider();
    const role = document.getElementById('register-role').value;

    try {
      const result = await signInWithPopup(auth, provider);
      const user = result.user;

      const userRef = doc(db, "users", user.uid);
      const userSnap = await getDoc(userRef);

      if (!userSnap.exists()) {
        await setDoc(userRef, {
          fullName: user.displayName || "Google User",
          country: "Global",
          city: "",
          languages: "",
          skills: "",
          experience: "",
          availability: role === 'worker' ? "OPEN TO WORK" : "HIRING",
          timezone: "",
          linkedin: "",
          portfolio: "",
          bio: "",
          email: user.email,
          role: role,
          createdAt: new Date().toISOString()
        });
      }

      showToast("Successfully signed in with Google!");
    } catch (error) {
      showToast("Google sign-in error: " + error.message, "error");
    }
  };

  window.forgotPassword = async () => {
    const email = document.getElementById('auth-email').value.trim();

    if (!email) {
      showToast("Please enter your email address in the Email field above first, then click 'Forgot Password?'.", "error");
      return;
    }

    try {
      await sendPasswordResetEmail(auth, email);
      showToast("Password reset email sent!<br>📧 Check your Inbox or Spam folder for the reset link.");
    } catch (error) {
      showToast("Error: " + error.message, "error");
    }
  };

  onAuthStateChanged(auth, async (user) => {
    const guestView = document.getElementById('gmt-guest-view');
    const verifyView = document.getElementById('gmt-verify-view');
    const workerDash = document.getElementById('gmt-worker-dashboard');
    const recruiterDash = document.getElementById('gmt-recruiter-dashboard');

    if (user) {
      guestView.classList.add('gmt-hidden');

      if (!user.emailVerified && !user.providerData.some(p => p.providerId === 'google.com')) {
        verifyView.classList.remove('gmt-hidden');
        workerDash.classList.add('gmt-hidden');
        recruiterDash.classList.add('gmt-hidden');
        return;
      }

      verifyView.classList.add('gmt-hidden');
      
      try {
        const userDoc = await getDoc(doc(db, "users", user.uid));
        if (userDoc.exists()) {
          const data = userDoc.data();
          const role = data.role;
          
          if (role === 'recruiter') {
            recruiterDash.classList.remove('gmt-hidden');
            workerDash.classList.add('gmt-hidden');

            document.getElementById('gmt-recruiter-name').innerText = data.fullName || "Recruiter";
            document.getElementById('gmt-recruiter-email').innerText = "📧 " + (data.email || user.email);
            document.getElementById('gmt-recruiter-country').innerText = "🌍 " + (data.country || "United States");
            document.getElementById('gmt-recruiter-role-badge').innerText = "🟣 HIRING / RECRUITING";

          } else if (role === 'worker' || role === 'admin') {
            workerDash.classList.remove('gmt-hidden');
            recruiterDash.classList.add('gmt-hidden');

            document.getElementById('gmt-user-name').innerText = data.fullName || "Job Seeker";
            document.getElementById('gmt-user-email').innerText = "📧 " + (data.email || user.email);
            document.getElementById('gmt-user-country').innerText = "🌍 " + (data.country || "Global");
            
            document.getElementById('gmt-user-city').innerText = data.city ? "🏙 " + data.city : "";
            document.getElementById('gmt-user-languages').innerText = data.languages ? "🗣 " + data.languages : "";
            document.getElementById('gmt-user-skills').innerText = data.skills ? "🛠 " + data.skills : "";
            document.getElementById('gmt-user-experience').innerText = data.experience ? "⭐ " + data.experience : "";
            document.getElementById('gmt-user-timezone').innerText = data.timezone ? "🕒 " + data.timezone : "";

            let linksHtml = "";
            if (data.linkedin) linksHtml += `<a href="${data.linkedin}" target="_blank" style="color: #0a66c2; text-decoration: none; margin-right: 15px;">🔗 LinkedIn Profile</a>`;
            if (data.portfolio) linksHtml += `<a href="${data.portfolio}" target="_blank" style="color: #0a66c2; text-decoration: none;">🌐 Portfolio / Website</a>`;
            document.getElementById('gmt-user-links').innerHTML = linksHtml;

            document.getElementById('gmt-user-role-badge').innerText = "🟢 " + (data.availability || "OPEN TO WORK");

            // Profile Completion Calculation (Total 6 Core Fields)
            let filledFields = 0;
            const totalFields = 6;
            if (data.fullName) filledFields++;
            if (data.country) filledFields++;
            if (data.languages) filledFields++;
            if (data.skills) filledFields++;
            if (data.experience) filledFields++;
            if (data.bio) filledFields++;

            const percentage = Math.round((filledFields / totalFields) * 100);
            document.getElementById('gmt-progress-text').innerText = percentage + "%";
            document.getElementById('gmt-progress-fill').style.width = percentage + "%";

          } else {
            guestView.classList.remove('gmt-hidden');
          }
        } else {
          guestView.classList.remove('gmt-hidden');
        }
      } catch (err) {
        console.error("Error loading profile:", err);
      }
    } else {
      guestView.classList.remove('gmt-hidden');
      verifyView.classList.add('gmt-hidden');
      workerDash.classList.add('gmt-hidden');
      recruiterDash.classList.add('gmt-hidden');
    }
  });

  // 1) Listener pou chèche epi voye PDF la lè itilizatè a chwazi l
  document.getElementById("gmt-resume-file").addEventListener("change", async function(){
    const file = this.files[0];
    if(!file){
      return;
    }
    if(file.type !== "application/pdf"){
      showToast("Please upload PDF only", "error");
      return;
    }

    const formData = new FormData();
    formData.append("resume", file);

    showToast("Uploading resume...");

    try {
      const response = await fetch("https://gaveai-backend.onrender.com/upload-resume", {
        method: "POST",
        body: formData
      });
      const data = await response.json();
      
      if(data.success){
        showToast("Resume uploaded successfully!");
        console.log("Resume URL:", data.url);
        
        // Sove URL PDF la nan Firestore pou itilizatè ki konekte a
        await saveResumeURL(data.url);

      } else {
        showToast("Upload failed", "error");
      }
    } catch(error) {
      console.error(error);
      showToast("Server connection error", "error");
    }
  });

  // Fonksyon pou anrejistre URL resume a nan Firestore
  async function saveResumeURL(url){
    const user = auth.currentUser;
    if(!user) return;
    
    try {
      await updateDoc(doc(db, "users", user.uid), {
        resumeURL: url,
        resumeUpdatedAt: new Date()
      });
      console.log("Resume URL saved to Firestore!");
      location.reload(); // Rafrechi pou lyen "View Resume" a parèt sou dashboard la
    } catch (err) {
      console.error("Error saving resume URL to Firestore:", err);
    }
  }
